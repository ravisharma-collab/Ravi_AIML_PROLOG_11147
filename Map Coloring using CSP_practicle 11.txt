% Colors
color(red).
color(green).
color(blue).

% Neighboring regions
neighbor(a, b).
neighbor(a, c).
neighbor(b, c).

% Different colors for neighbors
different(A, B) :-
    A \= B.

% Assign colors
color_map(A, B, C) :-
    color(A),
    color(B),
    color(C),
    different(A, B),
    different(A, C),
    different(B, C).
