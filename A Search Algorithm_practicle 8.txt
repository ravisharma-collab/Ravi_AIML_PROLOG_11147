astar(Start, Goal, Path, Cost) :-
    astar_search([[0,Start,[Start]]], Goal, Rev, Cost),
    reverse(Rev, Path).

astar_search([[Cost,Goal,Path]|_], Goal, Path, Cost).

astar_search([[G,Node,Path]|Rest], Goal, Result, Cost) :-
    findall(
        [F,Next,[Next|Path]],
        (edge(Node,Next,W),
         \+ member(Next,Path),
         NewG is G+W,
         h(Next,H),
         F is NewG+H),
        Children
    ),
    append(Rest,Children,Open),
    sort(Open,Sorted),
    astar_search(Sorted,Goal,Result,Cost).
