% Performance rules
performance(Name, topper) :-
    marks(Name, M),
    M >= 80.

performance(Name, average) :-
    marks(Name, M),
    M >= 40,
    M < 80.

performance(Name, fail) :-
    marks(Name, M),
    M < 40.
